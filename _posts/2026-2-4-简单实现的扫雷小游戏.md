## C实现

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <time.h>
#include <pthread.h>
#include <unistd.h>
#include <conio.h>
#include <windows.h>

typedef unsigned int u32;

enum {
    MINE_EMPTY = 0, /* 空地 */
    MINE_MINE,      /* 地雷 */
    MINE_MAX,
};

enum {
    KEY_UP = 72,
    KEY_DOWN = 80,
    KEY_LEFT = 75,
    KEY_RIGHT = 77,
};

typedef struct field_point_s {
    u32 x;
    u32 y;
    u32 type;
    u32 mine_num;
    bool is_open;
    bool is_flag;
} field_point_t;

typedef struct mine_s {
    u32 width;
    u32 hight;
    u32 mine_num;
    field_point_t **fields;
    bool **access;        /* 用于解锁空白的遍历 */
} mine_t;

mine_t g_mine;
u32 cur_x;
u32 cur_y;
bool is_game_over;
bool is_game_restart;
bool is_game_win;

u32 mine_get_feild_num(field_point_t **fields, u32 xbegin, u32 xend, u32 ybegin, u32 yend)
{
    u32 num = 0;
    u32 i;
    u32 j;

    for (i = xbegin; i <= xend; ++i) {
        for (j = ybegin; j <= yend; ++j) {
            if (fields[i][j].type == MINE_MINE) {
                num++;
            }
        }
    }

    return num;
}

int mine_calculate_num(field_point_t **fields, u32 width, u32 hight)
{
    u32 i;
    u32 j;
    u32 mine_cnt;
    u32 k, m;

    for (i = 0; i < width; i++) {
        for (j = 0; j < hight; j++) {
            if (fields[i][j].type == MINE_MINE) {
                continue;
            }

            if (i == 0) {
                if (j == 0) {
                    mine_cnt = mine_get_feild_num(fields, i, i+1, j, j+1);
                } else if (j == hight - 1) {
                    mine_cnt = mine_get_feild_num(fields, i, i+1, j-1, j);
                } else {
                    mine_cnt = mine_get_feild_num(fields, i, i+1, j-1, j+1);
                }
            } else if (i == width - 1) {
                if (j == 0) {
                    mine_cnt = mine_get_feild_num(fields, i-1, i, j, j+1);
                } else if (j == hight - 1) {
                    mine_cnt = mine_get_feild_num(fields, i-1, i, j-1, j);
                } else {
                    mine_cnt = mine_get_feild_num(fields, i-1, i, j-1, j+1);
                }
            } else {
                if (j == 0) {
                    mine_cnt = mine_get_feild_num(fields, i-1, i+1, j, j+1);
                } else if (j == hight - 1) {
                    mine_cnt = mine_get_feild_num(fields, i-1, i+1, j-1, j);
                } else {
                    mine_cnt = mine_get_feild_num(fields, i-1, i+1, j-1, j+1);
                }
            }

            fields[i][j].mine_num = mine_cnt;
        }
    }
}

int mine_construct(field_point_t **fields, u32 hight, u32 width, u32 mine_num)
{
    u32 i;
    u32 j;
    u32 rand_i, rand_j;

    if (fields == NULL || hight == 0 || width == 0 || mine_num == 0) {
        return -1;
    }

    /* init */
    for (i = 0; i < hight; i++) {
        for (j = 0; j < width; j++) {
            fields[i][j].x = i;
            fields[i][j].y = j;
            fields[i][j].type = MINE_EMPTY;
            fields[i][j].is_open = false;
            fields[i][j].is_flag = false;
        }
    }

    /* 重置随机数 */
    srand(time(NULL));

    for (i = 0; i < mine_num; ++i) {
        /* 随机生成一个坐标 */
        rand_i = rand() % hight;
        rand_j = rand() % width;
        if (fields[rand_i][rand_j].type == MINE_EMPTY) {
            fields[rand_i][rand_j].type = MINE_MINE;
        } else {
            i--;
        }
    }

    return 0;
}

void mine_open_zero(u32 x, u32 y)
{
    /* 将当前位置的周围的空白格子全部打开 */
    field_point_t **fields;
    u32 width;
    u32 hight;
    mine_t *mine;

    mine = &g_mine;

    if (mine->access[x][y] || mine->fields[x][y].is_open) {
        return;
    }

    fields = mine->fields;
    width = mine->width;
    hight = mine->hight;

    mine->access[x][y] = true;
    fields[x][y].is_open = true;
    if (fields[x][y].type == MINE_EMPTY && fields[x][y].mine_num == 0) {
        if (x > 0) {
            mine_open_zero(x - 1, y);
        }
        if (x + 1 < hight) {
            mine_open_zero(x + 1, y);
        }
        if (y > 0) {
            mine_open_zero(x, y - 1);
        }
        if (y + 1 < width) {
            mine_open_zero(x, y + 1);
        }
    }
}


void *convas_print(void *arg)
{
    u32 i;
    u32 j;
    mine_t *mine;
    field_point_t **fields;
    u32 width;
    u32 hight;
    u32 ab;
    bool a;

    mine = &g_mine;
    fields = mine->fields;
    width = mine->width;
    hight = mine->hight;
    is_game_restart = false;
    is_game_over = false;
    is_game_win = false;

    ab = 1;
    a = true;

    while (1) {
        if (is_game_over || is_game_win || is_game_restart) {
            break;
        }
        // printf("\033[H"); // 将光标移到屏幕左上角
        printf("\033[H"); // 如果需要清除后面内容也可以用这一句
        is_game_win = true;
        if (ab % 5 == 0) {
            if (a) a = false;
            else a = true;
        }

        for (i = 0; i < hight; i++) {
            for (j = 0; j < width; j++) {
                if (a && cur_x == i && cur_y == j) {
                    printf("@ ");
                    continue;
                }

                if (fields[i][j].is_flag) {
                    printf("F ");
                } else if (fields[i][j].is_open) {
                    if (fields[i][j].type == MINE_MINE) {
                        printf("* ");
                    } else {
                        printf("%u ", fields[i][j].mine_num);
                    }
                } else {
                    printf("- ");
                }

                if (!fields[i][j].is_open && !fields[i][j].is_flag) {
                    is_game_win = false;
                }
            }
            printf("\n");
        }

        printf("move: ↑↓←→\n");
        printf("open: space\n");
        printf("set flag: f\n");
        printf("exit: q\n");
        printf("restart: r\n");
        ab++;
        usleep(50000);
    }

    if (is_game_over) {
        printf("game over!\n");
    } else if (is_game_win) {
        printf("game win!\n");
    } else if (is_game_restart) {
        printf("game restart!\n");
    }

    return NULL;
}

void *game_start(void *arg)
{
    char key;
    pthread_t tid;
    int ret;
    mine_t *mine;
    field_point_t **fields;
    u32 width;
    u32 hight;
    u32 mine_num;
    int i;

    mine = &g_mine;
    fields = mine->fields;
    width = mine->width;
    hight = mine->hight;
    mine_num = mine->mine_num;

    ret = pthread_create(&tid, NULL, (void *)convas_print, NULL);
    if (ret != 0) {
        printf("pthread_create error: %d\n", ret);
        return NULL;
    }

    pthread_detach(tid);

    while (1) {
        /* 监听方向键，无需回车，立即响应 */
        key = _getch();

        switch (key) {
            case KEY_UP:
                if (cur_x > 0) {
                    cur_x--;
                }
                break;
            case KEY_DOWN:
                if (cur_x < hight - 1) {
                    cur_x++;
                }
                break;
            case KEY_LEFT:
                if (cur_y > 0) {
                    cur_y--;
                }
                break;
            case KEY_RIGHT:
                if (cur_y < width - 1) {
                    cur_y++;
                }
                break;
            case 'q':
                exit(0);
                break;
            case 'r':
                is_game_restart = true;
                sleep(1);
                mine_construct(fields, hight, width, mine_num);
                mine_calculate_num(fields, hight, width);
                ret = pthread_create(&tid, NULL, (void *)convas_print, NULL);
                if (ret != 0) {
                    printf("pthread_create error: %d\n", ret);
                    return NULL;
                }

                pthread_detach(tid);
                break;
            case 'f':
                if (!g_mine.fields[cur_x][cur_y].is_open) {
                    if (g_mine.fields[cur_x][cur_y].is_flag) {
                        g_mine.fields[cur_x][cur_y].is_flag = false;
                    } else {
                        g_mine.fields[cur_x][cur_y].is_flag = true;
                    }
                }
                break;
            case ' ':
                /* 插旗子的不能翻开 */
                if (g_mine.fields[cur_x][cur_y].is_flag) {
                    break;
                }
                if (g_mine.fields[cur_x][cur_y].type == MINE_MINE) {
                    is_game_over = true;
                } else {
                    if (g_mine.fields[cur_x][cur_y].mine_num == 0) {
                        for (i = 0; i < hight; ++i) {
                            memset(g_mine.access[i], 0, sizeof(bool) * width);
                        }
                        mine_open_zero(cur_x, cur_y);
                    }
                    g_mine.fields[cur_x][cur_y].is_open = true;
                }
                break;
            default:
                break;
        }
        usleep(50000);
    }
}

void run(void)
{
    pthread_t tid;
    int ret;

    cur_x = 0;
    cur_y = 0;
    is_game_over = false;


    ret = pthread_create(&tid, NULL, (void *)game_start, NULL);
    if (ret != 0) {
        printf("pthread_create error: %d\n", ret);
        return;
    }

    pthread_join(tid, NULL);
}

void init_window(void)
{
    // 获取标准输出句柄
    HANDLE hOut = GetStdHandle(STD_OUTPUT_HANDLE);
    if (hOut == INVALID_HANDLE_VALUE) {
        return;
    }

    // 获取当前控制台模式
    DWORD dwMode = 0;
    if (!GetConsoleMode(hOut, &dwMode)) {
        return;
    }

    // 启用对ANSI转义序列的支持
    dwMode |= ENABLE_VIRTUAL_TERMINAL_PROCESSING;
    if (!SetConsoleMode(hOut, dwMode)) {
        return;
    }
}

int main(void)
{
    field_point_t **fields;
    bool **access;
    u32 i;
    u32 j;
    u32 hight = 10;
    u32 width = 10;
    u32 mine_num = 10;

    init_window();

    fields = (field_point_t **)malloc(sizeof(field_point_t *) * hight);
    if (fields == NULL) {
        return -1;
    }

    memset(fields, 0, sizeof(field_point_t *) * hight);

    for (i = 0; i < hight; i++) {
        fields[i] = (field_point_t *)malloc(sizeof(field_point_t) * width);
        if (fields[i] == NULL) {
            goto free;
        }
        memset(fields[i], 0, sizeof(field_point_t) * width);
    }

    access = (bool **)malloc(sizeof(bool *) * hight);
    if (access == NULL) {
        goto free;
    }
    memset(access, 0, sizeof(bool *) * hight);

    for (i = 0; i < hight; i++) {
        access[i] = (bool *)malloc(sizeof(bool) * width);
        if (access[i] == NULL) {
            goto free;
        }
        memset(access[i], 0, sizeof(bool) * width);
    }

    mine_construct(fields, hight, width, mine_num);
    mine_calculate_num(fields, hight, width);

    g_mine.fields = fields;
    g_mine.hight = hight;
    g_mine.width = width;
    g_mine.mine_num = mine_num;
    g_mine.access = access;

    run();

    for (i = 0; i < hight; i++) {
        for (j = 0; j < width; j++) {
            printf("%d ", fields[i][j].type);
        }
        printf("\n");
    }

free:

    for (i = 0; i < hight; i++) {
        if (fields[i] != NULL) {
            free(fields[i]);
        } else {
            break;
        }
    }

    if (fields != NULL) {
        free(fields);
    }

    return 0;
}
```

## Python 实现

使用到了tk窗口，更贴近我们平常玩的扫雷，但是由于不会画旗子，就用问号代替了

```python
# -*- coding: gbk -*-
import tkinter as tk
from tkinter import messagebox
import random
import time

MINE_EMPTY = 0
MINE_MINE = 1

class FieldPoint:
    def __init__(self, x: int, y: int):
        self.x = x
        self.y = y
        self.type = MINE_EMPTY
        self.is_open = False
        self.is_flag = False
        self.mine_num = 0

class Mine:
    def __init__(self, height: int, width: int, mine_num: int):
        self.height = height
        self.width = width
        self.mine_num = mine_num
        self.fields = [
            [FieldPoint(i, j) for j in range(width)] for i in range(height)
        ]
        self.access = [[False] * width for _ in range(height)]

    def mine_get_field_num(self, xbegin: int, xend: int, ybegin: int, yend: int) -> int:
        num = 0
        for i in range(xbegin, xend + 1):
            for j in range(ybegin, yend + 1):
                if self.fields[i][j].type == MINE_MINE:
                    num += 1
        return num

    def mine_calculate_num(self):
        for i in range(self.height):
            for j in range(self.width):
                if self.fields[i][j].type == MINE_MINE:
                    continue
                xbegin = max(0, i - 1)
                xend = min(self.height - 1, i + 1)
                ybegin = max(0, j - 1)
                yend = min(self.width - 1, j + 1)
                mine_cnt = self.mine_get_field_num(xbegin, xend, ybegin, yend)
                self.fields[i][j].mine_num = mine_cnt

    def mine_construct(self):
        if self.height == 0 or self.width == 0 or self.mine_num == 0:
            return -1
        for i in range(self.height):
            for j in range(self.width):
                self.fields[i][j].x = i
                self.fields[i][j].y = j
                self.fields[i][j].type = MINE_EMPTY
                self.fields[i][j].is_open = False
                self.fields[i][j].is_flag = False
                self.fields[i][j].mine_num = 0

        random.seed(time.time())
        placed = 0
        while placed < self.mine_num:
            rand_i = random.randint(0, self.height - 1)
            rand_j = random.randint(0, self.width - 1)
            if self.fields[rand_i][rand_j].type == MINE_EMPTY:
                self.fields[rand_i][rand_j].type = MINE_MINE
                placed += 1

        self.mine_calculate_num()
        return 0

    def mine_open_zero(self, x: int, y: int):
        if self.access[x][y] or self.fields[x][y].is_open or self.fields[x][y].is_flag:
            return
        self.access[x][y] = True
        self.fields[x][y].is_open = True

        if self.fields[x][y].type == MINE_EMPTY and self.fields[x][y].mine_num == 0:
            if x > 0:
                self.mine_open_zero(x - 1, y)
            if x + 1 < self.height:
                self.mine_open_zero(x + 1, y)
            if y > 0:
                self.mine_open_zero(x, y - 1)
            if y + 1 < self.width:
                self.mine_open_zero(x, y + 1)

class MineGUI(tk.Tk):
    def __init__(self, height, width, mine_num):
        super().__init__()
        self.title("扫雷")
        self.mine = Mine(height, width, mine_num)
        self.mine.mine_construct()

        self.buttons = [[None for _ in range(width)] for _ in range(height)]
        restart_btn = tk.Button(self, text="重新开始", command=self.restart)
        restart_btn.grid(row=0, column=0, columnspan=width, sticky="we")

        # 创建按钮
        for i in range(height):
            for j in range(width):
                btn = tk.Button(self, width=2, height=1, font=("Arial", 14),
                                command=lambda x=i, y=j: self.left_click(x, y))
                btn.bind("<Button-3>", lambda e, x=i, y=j: self.right_click(x, y))
                btn.grid(row=i+1, column=j)
                self.buttons[i][j] = btn

    def restart(self):
        self.mine.mine_construct()
        self.mine.access = [[False] * self.mine.width for _ in range(self.mine.height)]
        for i in range(self.mine.height):
            for j in range(self.mine.width):
                btn = self.buttons[i][j]
                btn.config(text="", bg="SystemButtonFace", relief=tk.RAISED, state=tk.NORMAL, fg="black")
                field = self.mine.fields[i][j]
                field.is_open = False
                field.is_flag = False
        # 确保按钮显示更新
        self.update_buttons()

    def left_click(self, x, y):
        field = self.mine.fields[x][y]
        if field.is_flag or field.is_open:
            return
        if field.type == MINE_MINE:
            self.buttons[x][y].config(text="?", bg="red")
            self.game_over(False)
            return
        # 打开空白格子
        self.mine.access = [[False] * self.mine.width for _ in range(self.mine.height)]  # 重置访问
        self.mine.mine_open_zero(x, y)
        self.update_buttons()
        if self.check_win():
            self.game_over(True)

    def right_click(self, x, y):
        field = self.mine.fields[x][y]
        if field.is_open:
            return
        field.is_flag = not field.is_flag
        btn = self.buttons[x][y]
        if field.is_flag:
            btn.config(text="?", fg="red")
        else:
            btn.config(text="", fg="black")

    def update_buttons(self):
        for i in range(self.mine.height):
            for j in range(self.mine.width):
                field = self.mine.fields[i][j]
                btn = self.buttons[i][j]
                if field.is_open:
                    btn.config(relief=tk.SUNKEN, state=tk.DISABLED, bg="lightgrey")
                    if field.type == MINE_EMPTY:
                        if field.mine_num > 0:
                            btn.config(text=str(field.mine_num),
                                       fg=self.get_color(field.mine_num))
                        else:
                            btn.config(text="")
                    elif field.type == MINE_MINE:
                        btn.config(text="?", bg="red")
                else:
                    if not field.is_flag:
                        btn.config(text="", relief=tk.RAISED, state=tk.NORMAL, bg="SystemButtonFace")

    def get_color(self, num):
        colors = {
            1: "blue",
            2: "green",
            3: "red",
            4: "darkblue",
            5: "darkred",
            6: "cyan",
            7: "black",
            8: "gray"
        }
        return colors.get(num, "black")

    def game_over(self, win):
        # 打开所有雷
        for i in range(self.mine.height):
            for j in range(self.mine.width):
                field = self.mine.fields[i][j]
                btn = self.buttons[i][j]
                if field.type == MINE_MINE:
                    btn.config(text="?", bg="red")
                btn.config(state=tk.DISABLED)
        if win:
            messagebox.showinfo("游戏结束", "恭喜你，赢了！")
        else:
            messagebox.showinfo("游戏结束", "很遗憾，你踩到雷了！")

    def check_win(self):
        # 判断所有非雷格子是否都打开了
        for i in range(self.mine.height):
            for j in range(self.mine.width):
                field = self.mine.fields[i][j]
                if field.type == MINE_EMPTY and not field.is_open:
                    return False
        return True

if __name__ == "__main__":
    height = 10
    width = 10
    mine_num = 15
    app = MineGUI(height, width, mine_num)
    app.mainloop()
```
