## 第一题

最长的相同字符前缀，"flower", "flow", "flight"的相同字符前缀是"fl"，这种类似

```c
#include <stdio.h>
#include <string.h>

void longest_common_prefix(char* str_list[], char* res, size_t res_size, int str_cnt)
{
    int i;
    int index;
    char new_char;
    int min_length;

    if (str_cnt == 0) {
        res[0] = '\0';
        return;
    }

    index = 0;
    if (str_cnt == 1) {
        snprintf(res, res_size, "%s", str_list[0]);
        return;
    }

    min_length = strlen(str_list[0]);
    for (int i = 1; i < str_cnt; ++i) {
        int length = strlen(str_list[i]);
        if (length < min_length) {
            min_length = length;
        }
    }

    while (index < min_length) {
        new_char = str_list[0][index];
        for (i = 1; i < str_cnt; ++i) {
            if (new_char != str_list[i][index]) {
                res[index] = '\0';
                return;
            }
        }

        res[index++] = new_char;
    }

    res[index] = '\0';
}

int main(void)
{
    char* strs[3] = {"flower", "flow", "flight"};
    char res[10] = {0};

    longest_common_prefix(strs, res, sizeof(res), 3);

    printf("%s", res);

    return 0;
}
```
